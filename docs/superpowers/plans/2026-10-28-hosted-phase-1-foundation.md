# Hosted Resume Tailor — Phase 1 (Foundation) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the local, single-machine Resume Tailor into a signed-in, Postgres-backed web app that runs in one Docker container on Fly.io, so Edward can use his imported bank on the live site.

**Architecture:** Repositories keep their method signatures but are keyed by the signed-in user's id instead of a profile slug, and are backed by Postgres through Drizzle (PGlite when `DATABASE_URL` is unset, so dev and tests need no services). Better Auth provides Google sign-in and email magic links; the session is read on the server, the client never sends an identity. One Dockerfile bundles Node and a minimal TeX Live; migrations run when the server boots.

**Tech Stack:** Next.js 16.4 (App Router, `proxy.ts`), React 19, TypeScript, zod 4, Drizzle ORM 0.45 + drizzle-kit 0.31, `pg` 8, `@electric-sql/pglite` 0.5, Better Auth 1.7 + `@better-auth/drizzle-adapter`, Resend 6, Vitest 5, Playwright 1.64, TeX Live via `install-tl`, Fly.io.

Spec: `docs/superpowers/specs/2026-10-28-hosted-resume-tailor-design.md`. Phases 2 (payments, limits, legal, landing) and 3 (ads) get their own plans.

**Conventions for every task**

```ts
import path from "node:path";
import type { NodePgDatabase } from "drizzle-orm/node-postgres";
import { migrate as migratePg } from "drizzle-orm/node-postgres/migrator";
import type { PgDatabase, PgQueryResultHKT } from "drizzle-orm/pg-core";
import type { PgliteDatabase } from "drizzle-orm/pglite";
import { migrate as migratePglite } from "drizzle-orm/pglite/migrator";
import type * as schema from "./schema";

export type Db = PgDatabase<PgQueryResultHKT, typeof schema>;
export type Connection =
  | { kind: "pg"; db: NodePgDatabase<typeof schema> }
  | { kind: "pglite"; db: PgliteDatabase<typeof schema> };

export const MIGRATIONS_FOLDER = path.join(process.cwd(), "drizzle");

export async function runMigrations(connection: Connection): Promise<void> {
  if (connection.kind === "pg") await migratePg(connection.db, { migrationsFolder: MIGRATIONS_FOLDER });
  else await migratePglite(connection.db, { migrationsFolder: MIGRATIONS_FOLDER });
}
```

`src/db/client.ts`:

```ts
import path from "node:path";
import { drizzle as drizzlePg } from "drizzle-orm/node-postgres";
import { drizzle as drizzlePglite } from "drizzle-orm/pglite";
import { Pool } from "pg";
import { dataRoot } from "@/lib/data-root";
import { type Connection, type Db, runMigrations } from "./migrate";
import * as schema from "./schema";

// Next dev re-evaluates modules on hot reload; the pool/PGlite instance must survive that.
const cache = globalThis as unknown as { __resumeDb?: { connection: Connection; ready: Promise<void> } };

function connect(): Connection {
  const url = process.env.DATABASE_URL;
  if (url) return { kind: "pg", db: drizzlePg({ client: new Pool({ connectionString: url, max: 5 }), schema }) };
  return { kind: "pglite", db: drizzlePglite({ connection: { dataDir: path.join(dataRoot(), "pglite") }, schema }) };
}

function init() {
  if (!cache.__resumeDb) {
    const connection = connect();
    const ready = process.env.MIGRATE_ON_START === "0" ? Promise.resolve() : runMigrations(connection);
    cache.__resumeDb = { connection, ready };
  }
  return cache.__resumeDb;
}

/** The process-wide database, migrated before it is handed out. */
export async function getDb(): Promise<Db> {
  const { connection, ready } = init();
  await ready;
  return connection.db;
}

/** For code that needs the instance synchronously (the auth adapter); migrations may still be running. */
export function getDbSync(): Db {
  return init().connection.db;
}
```

`src/bank/repository.ts`:

```ts
import { eq } from "drizzle-orm";
import type { Db } from "@/db/migrate";
import { banks } from "@/db/schema";
import { AppError } from "@/lib/errors";
import { type Bank, BankSchema, emptyBank } from "./schema";

/** One bank per user, stored as validated JSON. `db` is a getter so the connection opens lazily. */
export class BankRepository {
  constructor(private readonly db: () => Promise<Db>) {}

  async createEmpty(userId: string, name: string): Promise<void> {
    const db = await this.db();
    await db.insert(banks).values({ userId, data: emptyBank(name) }).onConflictDoNothing();
  }

  async read(userId: string): Promise<Bank> {
    const db = await this.db();
    const [row] = await db.select({ data: banks.data }).from(banks).where(eq(banks.userId, userId));
    if (!row) throw new AppError("No bank for this account yet", 404);
    const parsed = BankSchema.safeParse(row.data);
    if (!parsed.success) throw new AppError("Stored bank does not match the schema", 500, parsed.error.issues);
    return parsed.data;
  }

  async write(userId: string, bank: unknown): Promise<Bank> {
    const parsed = BankSchema.safeParse(bank);
    if (!parsed.success) throw new AppError("Bank does not match the schema", 400, parsed.error.issues);
    const db = await this.db();
    await db
      .insert(banks)
      .values({ userId, data: parsed.data })
      .onConflictDoUpdate({ target: banks.userId, set: { data: parsed.data, updatedAt: new Date() } });
    return parsed.data;
  }
}
```

`src/applications/repository.ts`:

```ts
import { and, desc, eq, isNotNull, sql } from "drizzle-orm";
import type { Db } from "@/db/migrate";
import { applications } from "@/db/schema";
import { AppError } from "@/lib/errors";
import { assertApplicationId } from "./ids";
import { type Application, ApplicationSchema, type ApplicationSummary } from "./schema";

export class ApplicationRepository {
  constructor(private readonly db: () => Promise<Db>) {}

  async list(userId: string): Promise<ApplicationSummary[]> {
    const db = await this.db();
    const rows = await db
      .select({
        id: applications.id,
        createdAt: sql<string>`${applications.data}->>'createdAt'`,
        company: sql<string>`${applications.data}->>'company'`,
        roleTitle: sql<string>`coalesce(${applications.data}->>'roleTitle', '')`,
        model: sql<string>`${applications.data}->>'model'`,
        hasPdf: isNotNull(applications.pdf),
      })
      .from(applications)
      .where(eq(applications.userId, userId))
      .orderBy(desc(sql`${applications.data}->>'createdAt'`), desc(applications.id));
    return rows.map((r) => ({ ...r, hasPdf: Boolean(r.hasPdf) }));
  }

  async get(userId: string, id: string): Promise<Application> {
    const db = await this.db();
    const [row] = await db
      .select({ data: applications.data })
      .from(applications)
      .where(and(eq(applications.userId, userId), eq(applications.id, assertApplicationId(id))));
    if (!row) throw new AppError(`Application "${id}" not found`, 404);
    const parsed = ApplicationSchema.safeParse(row.data);
    if (!parsed.success) throw new AppError("Stored application does not match the schema", 500, parsed.error.issues);
    return parsed.data;
  }

  async save(userId: string, application: unknown): Promise<Application> {
    const parsed = ApplicationSchema.safeParse(application);
    if (!parsed.success) throw new AppError("Application does not match the schema", 400, parsed.error.issues);
    const db = await this.db();
    await db
      .insert(applications)
      .values({ userId, id: parsed.data.id, data: parsed.data })
      .onConflictDoUpdate({
        target: [applications.userId, applications.id],
        set: { data: parsed.data, updatedAt: new Date() },
      });
    return parsed.data;
  }

  async delete(userId: string, id: string): Promise<void> {
    const db = await this.db();
    await db.delete(applications).where(and(eq(applications.userId, userId), eq(applications.id, assertApplicationId(id))));
  }

  async savePdf(userId: string, id: string, pdf: Buffer, pages: number): Promise<void> {
    const db = await this.db();
    const updated = await db
      .update(applications)
      .set({ pdf, pdfPages: pages, updatedAt: new Date() })
      .where(and(eq(applications.userId, userId), eq(applications.id, assertApplicationId(id))))
      .returning({ id: applications.id });
    if (updated.length === 0) throw new AppError(`Application "${id}" not found`, 404);
  }

  async readPdf(userId: string, id: string): Promise<{ pdf: Buffer; pages: number } | null> {
    const db = await this.db();
    const [row] = await db
      .select({ pdf: applications.pdf, pages: applications.pdfPages })
      .from(applications)
      .where(and(eq(applications.userId, userId), eq(applications.id, assertApplicationId(id))));
    if (!row?.pdf) return null;
    return { pdf: Buffer.from(row.pdf), pages: row.pages ?? 0 };
  }
}
```

```ts
import { mkdir, writeFile } from "node:fs/promises";
import path from "node:path";
import { Resend } from "resend";
import { dataRoot } from "./data-root";
import { env } from "./env";

/** Sends the sign-in link; without a Resend key it is written to data/magic-links/<email>.txt and logged. */
export async function sendMagicLinkEmail(email: string, url: string): Promise<void> {
  if (env.RESEND_API_KEY) {
    const resend = new Resend(env.RESEND_API_KEY);
    await resend.emails.send({
      from: env.EMAIL_FROM,
      to: email,
      subject: "Your Resume Tailor sign-in link",
      text: `Click to sign in (valid for 5 minutes):\n\n${url}\n\nIf you did not ask for this, ignore this email.`,
    });
    return;
  }
  if (env.NODE_ENV === "production") throw new Error("RESEND_API_KEY is not set");
  const dir = path.join(dataRoot(), "magic-links");
  await mkdir(dir, { recursive: true });
  await writeFile(path.join(dir, `${email.toLowerCase()}.txt`), url);
  console.log(`[dev] magic link for ${email}: ${url}`);
}
```

`src/proxy.ts`:

```ts
import { getSessionCookie } from "better-auth/cookies";
import { type NextRequest, NextResponse } from "next/server";

const PUBLIC = [/^\/signin(\/|$)/, /^\/api\/auth\//, /^\/api\/health$/, /^\/privacy$/, /^\/terms$/];

function sameOrigin(request: NextRequest): boolean {
  const origin = request.headers.get("origin");
  if (origin === null) return true; // top-level navigations and non-browser clients send none
  try {
    return new URL(origin).host === request.headers.get("host");
  } catch {
    return false;
  }
}

export function proxy(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const isApi = pathname.startsWith("/api/");
  // CSRF: a browser request from another site carries its Origin, and is refused.
  if (isApi && !sameOrigin(request)) {
    return NextResponse.json({ error: "Cross-origin request refused" }, { status: 403 });
  }
  if (PUBLIC.some((re) => re.test(pathname))) return NextResponse.next();
  // Optimistic check only (a cookie can be forged); requireUser and the (app) layout verify for real.
  if (!getSessionCookie(request)) {
    if (isApi) return NextResponse.json({ error: "Sign in to continue" }, { status: 401 });
    const signin = new URL("/signin", request.url);
    signin.searchParams.set("next", pathname + request.nextUrl.search);
    return NextResponse.redirect(signin);
  }
  return NextResponse.next();
}

export const config = { matcher: ["/((?!_next/|favicon\\.ico|.*\\.[a-z0-9]+$).*)"] };
```

- [ ] **Step 2: Implement**

```ts
/** A tiny concurrency limiter: jobs beyond `limit` wait in FIFO order. */
export function createQueue(limit: number) {
  let active = 0;
  const waiting: (() => void)[] = [];
  const release = () => {
    active--;
    waiting.shift()?.();
  };
  return {
    async run<T>(job: () => Promise<T>): Promise<T> {
      if (active >= limit) await new Promise<void>((resolve) => waiting.push(resolve));
      active++;
      try {
        return await job();
      } finally {
        release();
      }
    },
  };
}

export const compileQueue = createQueue(Number(process.env.COMPILE_CONCURRENCY ?? 2));
```
